/**
 * REFERENCE COPY — not imported by anything. Rename to .ts to use in a new project.
 * Production source: lib/prototype/camera.ts + lib/prototype/sceneConfig.ts
 *
 * THE PORTAL-SQUARE CAMERA
 *
 * The camera is not a 3D position. It is a rectangle on screen.
 *
 * Each world declares one "portal square": a real object in that world with a design-space centre
 * and a side length at rest. The camera state is where that square currently sits (px, py) and how
 * big it is (logS). Both worlds are described by the same three numbers, which is what makes a
 * threshold crossing between them one continuous move.
 *
 * See: shared/CAMERA_GRAMMAR.md, opening/OPENING_PORTAL_SKILL.md
 */

// ─────────────────────────────────────────────────────────────── design space

/**
 * A fixed virtual canvas that the viewport is cover-fitted to. Every coordinate in the system is in
 * these units, so nothing in the animation depends on the device resolution.
 */
export const DESIGN = { w: 1920, h: 1080 } as const;

/** centre of the design frame */
export const VC = { x: DESIGN.w / 2, y: DESIGN.h / 2 };

export type WorldId = 'A' | 'B';

export interface PortalDef {
  /** design-space centre of the square when the camera is at rest */
  cx: number;
  cy: number;
  /** side length in design units at rest */
  size: number;
}

/**
 * Derive the source world's portal from the motif's own geometry — never place it independently,
 * or the two will drift apart.
 *
 *   const MOTIF = { x: 1025, y: 225, cell: 22, n: 25 };
 *   A: { cx: MOTIF.x + 3.5 * MOTIF.cell, cy: MOTIF.y + 3.5 * MOTIF.cell, size: 5 * MOTIF.cell }
 */
export const PORTAL: Record<WorldId, PortalDef> = {
  A: { cx: 1102, cy: 302, size: 110 },
  B: { cx: 620, cy: 400, size: 360 },
};

// ─────────────────────────────────────────────────────────────── state

/** The camera channels. In a real film these live on a much wider flat state object. */
export interface CameraState {
  /** the portal square's centre on screen, in design units */
  px: number;
  py: number;
  /** LOG of its side length — always tween the log, never the raw size */
  logS: number;
  /** 0 = source world, 1 = destination. Switched only while the square covers the viewport. */
  world: number;
}

export function initialCameraState(): CameraState {
  return {
    px: PORTAL.A.cx,
    py: PORTAL.A.cy,
    logS: Math.log(PORTAL.A.size * 0.92),
    world: 0,
  };
}

// ─────────────────────────────────────────────────────────────── camera

/**
 * 2.5D perspective camera derived from the portal square.
 *
 * At rest (the square at its design position and size) every layer is drawn at design scale.
 * A layer at relative depth d is (1 + d) times further away than the portal plane, so it scales and
 * pans less (d > 0) or more (d < 0).
 */
export interface Camera {
  /** distance to the portal plane; 1 at rest */
  d: number;
  camX: number;
  camY: number;
}

export function cameraFor(world: WorldId, st: CameraState): Camera {
  const p = PORTAL[world];
  const s = Math.exp(st.logS);
  const d = p.size / s;
  return {
    d,
    camX: p.cx - (st.px - VC.x) * d,
    camY: p.cy - (st.py - VC.y) * d,
  };
}

/**
 * "Look at world point (cx, cy) at zoom z" expressed as the portal square the engine already uses.
 * This is how named shots in a geometry file become camera state.
 */
export function lookAt(world: WorldId, cx: number, cy: number, z: number) {
  const p = PORTAL[world];
  return {
    px: VC.x + (p.cx - cx) * z,
    py: VC.y + (p.cy - cy) * z,
    logS: Math.log(p.size * z),
  };
}

// ─────────────────────────────────────────────────────────────── layers

export interface LayerXf {
  /** scale factor (design → design-screen) */
  m: number;
  /** translation: screen = p * m + t */
  tx: number;
  ty: number;
  /** false when the layer is behind the camera */
  ok: boolean;
}

/**
 * Written into a reused object rather than returned fresh: this runs for every plate, prop and
 * particle, every frame.
 */
export function layerTransform(cam: Camera, depth: number, out: LayerXf): LayerXf {
  const den = cam.d + depth;
  if (den <= 0.004) {
    out.ok = false;
    out.m = 0;
    return out;
  }
  out.ok = true;
  out.m = (1 + depth) / den;
  out.tx = VC.x + (-VC.x * depth - cam.camX) / den;
  out.ty = VC.y + (-VC.y * depth - cam.camY) / den;
  return out;
}

// ─────────────────────────────────────────────────────────────── viewport

/** Maps design-screen coordinates to device pixels (the equivalent of object-fit: cover). */
export interface Viewport {
  vw: number;
  vh: number;
  /** scale */
  b: number;
  /** offset */
  ox: number;
  oy: number;
}

export function viewportFor(vw: number, vh: number): Viewport {
  const b = Math.max(vw / DESIGN.w, vh / DESIGN.h);
  return { vw, vh, b, ox: vw / 2 - VC.x * b, oy: vh / 2 - VC.y * b };
}

/** design-screen point → device px */
export const toScreen = (vp: Viewport, x: number, y: number) => ({
  x: x * vp.b + vp.ox,
  y: y * vp.b + vp.oy,
});

// ─────────────────────────────────────────────────────────────── the cover predicate

/**
 * True when the portal square fully contains the visible viewport.
 *
 * The world switch is only legal while this is true — then both worlds render an identical frame
 * and the switch is invisible. Verify it at the switch time for every supported aspect ratio; do
 * not rely on timing alone.
 */
export function portalCovers(st: CameraState, vp: Viewport): boolean {
  const half = Math.exp(st.logS) / 2;
  const hw = vp.vw / 2 / vp.b;
  const hh = vp.vh / 2 / vp.b;
  return (
    st.px - half <= VC.x - hw &&
    st.px + half >= VC.x + hw &&
    st.py - half <= VC.y - hh &&
    st.py + half >= VC.y + hh
  );
}

// ─────────────────────────────────────────────────────────────── plate placement

export interface PlateDef {
  id: string;
  world: WorldId | 'view';
  src: string;
  /** design-space top-left */
  x: number;
  y: number;
  /** size in design units */
  w: number;
  h: number;
  /** depth relative to the world's portal plane; < 0 is in front and can pass the camera */
  depth: number;
  blend?: 'screen' | 'soft-light';
  opacity?: number;
}

/**
 * Lay a plate's <img> out at the IMAGE'S OWN pixel size and return the extra scale that maps it to
 * its design size.
 *
 * A plate laid out at its design size (which can be tens of thousands of units) forces the browser
 * to keep a composited layer of that size; during a fast pull-back it cannot rasterise the tiles in
 * time and the layers underneath flash through. Laid out at pixel size, each layer is only as large
 * as its image. What is drawn on screen is identical.
 */
export function fitPlate(img: HTMLImageElement, def: { w: number; h: number }): [number, number] {
  const nw = img.naturalWidth;
  const nh = img.naturalHeight;
  if (!nw || !nh) return [1, 1];
  if (img.dataset.fit !== `${nw}x${nh}`) {
    img.style.width = `${nw}px`;
    img.style.height = `${nh}px`;
    img.dataset.fit = `${nw}x${nh}`;
  }
  return [def.w / nw, def.h / nh];
}

/** Per-frame placement of one plate. Only transform and opacity are ever written. */
export function placePlate(
  img: HTMLImageElement,
  def: PlateDef,
  cam: Camera,
  vp: Viewport,
  xf: LayerXf,
  visible: boolean,
): void {
  layerTransform(cam, def.depth, xf);
  let opacity = def.opacity ?? 1;
  let show = visible && xf.ok;
  // fade layers in front of the camera plane as they approach the lens
  if (show && def.depth < 0) opacity *= Math.min(1, Math.max(0, (7 - xf.m) / 3.2));
  if (!show || opacity <= 0.001) {
    if (img.style.visibility !== 'hidden') img.style.visibility = 'hidden';
    return;
  }
  const k = xf.m * vp.b;
  const [fx, fy] = fitPlate(img, def);
  const x = (def.x * xf.m + xf.tx) * vp.b + vp.ox;
  const y = (def.y * xf.m + xf.ty) * vp.b + vp.oy;
  if (img.style.visibility !== 'visible') img.style.visibility = 'visible';
  img.style.opacity = String(opacity);
  img.style.transform = `translate3d(${x}px, ${y}px, 0) scale(${k * fx}, ${k * fy})`;
}
